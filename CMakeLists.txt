cmake_minimum_required(VERSION 3.24)
project(OVMeshDRpp VERSION 0.4.0 LANGUAGES C CXX)
include(cmake/VerifyVendored.cmake)
include(cmake/LicenseNotices.cmake)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OVMESH_BUILD_DESKTOP "Build the native desktop application" ON)
option(OVMESH_ENABLE_HACKRF "Build direct HackRF receive support when available" ON)
option(OVMESH_ENABLE_RTLSDR "Build direct RTL-SDR receive support using the reviewed driver and libusb" ON)
option(OVMESH_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(OVMESH_BUILD_IQ_LAB "Build the explicitly authorized temporary IQ developer tool (Mac only)" OFF)

find_package(Threads REQUIRED)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.5.8 REQUIRED COMPONENTS Crypto)
if(NOT OPENSSL_VERSION STREQUAL "3.5.8")
  message(FATAL_ERROR "This prototype was reviewed with OpenSSL 3.5.8. Supply its reviewed local OPENSSL_ROOT_DIR; review and record dependency changes before changing this baseline.")
endif()
add_library(ovmesh_sqlite STATIC third_party/sqlite/sqlite3.c)
target_include_directories(ovmesh_sqlite PUBLIC third_party/sqlite)
target_compile_definitions(ovmesh_sqlite PUBLIC SQLITE_OMIT_LOAD_EXTENSION=1
  SQLITE_DQS=0 SQLITE_TRUSTED_SCHEMA=0 SQLITE_THREADSAFE=1
  SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_OMIT_JSON=1)
target_link_libraries(ovmesh_sqlite PRIVATE Threads::Threads)
# Pinned, allocation-free schema decoder; no generator or Python runtime is
# executed by the application build. Keep these ABI-affecting flags identical
# across the runtime, generated descriptors, and all consumers.
add_library(ovmesh_nanopb STATIC third_party/nanopb/pb_common.c third_party/nanopb/pb_decode.c)
target_include_directories(ovmesh_nanopb PUBLIC third_party/nanopb)
target_compile_definitions(ovmesh_nanopb PUBLIC PB_BUFFER_ONLY=1 PB_VALIDATE_UTF8=1
  PB_NO_ERRMSG=1 PB_MESSAGE_NESTING_MAX=8)
add_library(ovmesh_meshtastic_schema STATIC
  third_party/meshtastic/generated/meshtastic/config.pb.c
  third_party/meshtastic/generated/meshtastic/mesh.pb.c
  third_party/meshtastic/generated/meshtastic/telemetry.pb.c)
target_include_directories(ovmesh_meshtastic_schema PUBLIC third_party/meshtastic/generated)
target_link_libraries(ovmesh_meshtastic_schema PUBLIC ovmesh_nanopb)
find_package(PkgConfig QUIET)
if(OVMESH_ENABLE_HACKRF AND PkgConfig_FOUND)
  pkg_check_modules(HACKRF QUIET libhackrf)
endif()
set(OVMESH_HAVE_HACKRF_LIBRARY FALSE)
if(OVMESH_ENABLE_HACKRF)
  # Use actual library/header paths. Some host libhackrf.pc files embed an old
  # libusb header directory that the public HackRF interface does not require.
  find_path(HACKRF_INCLUDE_DIR NAMES libhackrf/hackrf.h hackrf.h HINTS ${HACKRF_INCLUDE_DIRS} PATH_SUFFIXES libhackrf)
  find_library(HACKRF_LIBRARY NAMES hackrf HINTS ${HACKRF_LIBRARY_DIRS})
  if(HACKRF_INCLUDE_DIR AND HACKRF_LIBRARY)
    add_library(hackrf_external UNKNOWN IMPORTED)
    set_target_properties(hackrf_external PROPERTIES IMPORTED_LOCATION "${HACKRF_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${HACKRF_INCLUDE_DIR}")
    set(OVMESH_HAVE_HACKRF_LIBRARY TRUE)
  endif()
endif()

# Bounded waveform discovery is independent of payload decoding and storage.
set(OVMESH_HAVE_RTLSDR_LIBRARY FALSE)
if(OVMESH_ENABLE_RTLSDR)
  if(PkgConfig_FOUND)
    pkg_check_modules(RTLUSB QUIET IMPORTED_TARGET libusb-1.0)
  endif()
  if(TARGET PkgConfig::RTLUSB)
    set(OVMESH_RTLUSB_TARGET PkgConfig::RTLUSB)
  else()
    find_path(RTLUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0)
    find_library(RTLUSB_LIBRARY NAMES usb-1.0 libusb-1.0)
    if(RTLUSB_INCLUDE_DIR AND RTLUSB_LIBRARY)
      add_library(rtlusb_external UNKNOWN IMPORTED)
      set_target_properties(rtlusb_external PROPERTIES IMPORTED_LOCATION "${RTLUSB_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${RTLUSB_INCLUDE_DIR}")
      set(OVMESH_RTLUSB_TARGET rtlusb_external)
    endif()
  endif()
  if(OVMESH_RTLUSB_TARGET)
    # Compile only the reviewed USB/tuner library. No upstream CMake execution,
    # utility binaries, rtl_tcp, udev installation or network acquisition.
    add_library(ovmesh_rtlsdr STATIC third_party/rtlsdr/src/librtlsdr.c
      third_party/rtlsdr/src/tuner_e4k.c third_party/rtlsdr/src/tuner_fc0012.c
      third_party/rtlsdr/src/tuner_fc0013.c third_party/rtlsdr/src/tuner_fc2580.c
      third_party/rtlsdr/src/tuner_r82xx.c)
    target_include_directories(ovmesh_rtlsdr PUBLIC third_party/rtlsdr/include)
    target_compile_definitions(ovmesh_rtlsdr PUBLIC rtlsdr_STATIC=1)
    target_link_libraries(ovmesh_rtlsdr PRIVATE ${OVMESH_RTLUSB_TARGET} Threads::Threads)
    set(OVMESH_HAVE_RTLSDR_LIBRARY TRUE)
  else()
    message(STATUS "RTL-SDR support unavailable: install/provide libusb development files, or disable OVMESH_ENABLE_RTLSDR.")
  endif()
endif()

add_library(ovmesh_discovery_experiment STATIC src/lora_discovery.cpp src/discovery_channelizer.cpp src/discovery_worker.cpp)
target_include_directories(ovmesh_discovery_experiment PUBLIC src)
target_link_libraries(ovmesh_discovery_experiment PUBLIC Threads::Threads)
if(NOT MSVC)
  target_compile_options(ovmesh_discovery_experiment PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(OVMESH_SANITIZERS AND NOT MSVC)
  target_compile_options(ovmesh_discovery_experiment PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ovmesh_discovery_experiment PUBLIC -fsanitize=address,undefined)
endif()

add_library(ovmesh_core src/phy.cpp src/phy_codec.cpp src/protocol.cpp src/engine.cpp src/storage.cpp src/report.cpp src/gps.cpp src/spectrum.cpp src/bursts.cpp src/file_chooser.cpp src/gps_discovery.cpp src/preferences.cpp src/licenses.cpp src/desktop_assets.cpp)
target_include_directories(ovmesh_core PUBLIC include PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ovmesh_core PUBLIC OpenSSL::Crypto ovmesh_sqlite ovmesh_meshtastic_schema Threads::Threads)
target_link_libraries(ovmesh_core PRIVATE ovmesh_discovery_experiment)
if(WIN32)
  target_link_libraries(ovmesh_core PRIVATE Advapi32 SetupAPI)
elseif(APPLE)
  target_link_libraries(ovmesh_core PRIVATE "-framework IOKit" "-framework CoreFoundation")
endif()
target_compile_definitions(ovmesh_core PUBLIC OVMESH_VERSION="${PROJECT_VERSION}")
if(OVMESH_HAVE_RTLSDR_LIBRARY)
  target_compile_definitions(ovmesh_core PRIVATE OVMESH_HAVE_RTLSDR=1)
  target_link_libraries(ovmesh_core PRIVATE ovmesh_rtlsdr)
endif()
if(OVMESH_HAVE_HACKRF_LIBRARY)
  target_compile_definitions(ovmesh_core PRIVATE OVMESH_HAVE_HACKRF=1)
  target_link_libraries(ovmesh_core PRIVATE hackrf_external)
else()
  message(STATUS "HackRF support disabled/unavailable. Synthetic reception and saved analysis remain available.")
endif()
if(MSVC)
  target_compile_options(ovmesh_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(ovmesh_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(OVMESH_SANITIZERS AND NOT MSVC)
  target_compile_options(ovmesh_sqlite PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(ovmesh_nanopb PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(ovmesh_meshtastic_schema PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(ovmesh_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ovmesh_core PUBLIC -fsanitize=address,undefined)
endif()

add_executable(ovmesh-cli src/main.cpp)
target_link_libraries(ovmesh-cli PRIVATE ovmesh_core)

if(OVMESH_BUILD_DESKTOP)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  # Keep the first Linux build on X11/XWayland; native Wayland is a later,
  # separately inventoried optional backend, not an implicit build expansion.
  set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/glfw EXCLUDE_FROM_ALL)
  find_package(OpenGL REQUIRED)
  add_library(ovmesh_imgui STATIC
    third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
    third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
    third_party/imgui/backends/imgui_impl_glfw.cpp
    third_party/imgui/backends/imgui_impl_opengl3.cpp)
  target_include_directories(ovmesh_imgui PUBLIC third_party/imgui)
  target_compile_definitions(ovmesh_imgui PUBLIC IMGUI_DISABLE_DEFAULT_SHELL_FUNCTIONS)
  target_link_libraries(ovmesh_imgui PUBLIC glfw OpenGL::GL)
  if(APPLE)
    add_executable(OVMeshDRpp MACOSX_BUNDLE src/main.cpp src/ui.cpp)
    set_target_properties(OVMeshDRpp PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "org.ovmesh.survey" MACOSX_BUNDLE_BUNDLE_NAME "OVMeshDRpp" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  else()
    add_executable(OVMeshDRpp src/main.cpp src/ui.cpp)
  endif()
  target_compile_definitions(OVMeshDRpp PRIVATE OVMESH_DESKTOP=1)
  target_link_libraries(OVMeshDRpp PRIVATE ovmesh_core ovmesh_imgui)
  install(TARGETS OVMeshDRpp BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
install(TARGETS ovmesh-cli RUNTIME DESTINATION bin)
